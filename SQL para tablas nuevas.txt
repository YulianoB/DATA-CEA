Excelente. Esa es una muy buena práctica: guardar cada cambio nuevo como un archivo SQL de respaldo/migración, para después aplicarlo con calma a producción.

Te recomiendo crear una carpeta dentro del proyecto:

cd C:\mi-proyecto
mkdir supabase
mkdir supabase\migrations

Y guardar ahí los SQL nuevos.

1) Primer backup SQL: tablas de matrículas/convenios

Crea este archivo:

C:\mi-proyecto\supabase\migrations\001_matriculas_convenios.sql

Pega esto dentro:

-- ======================================================
-- DATA CEA - Migración 001
-- Módulo Matrículas / Convenios
-- Tablas nuevas para aplicar posteriormente en producción
-- SIN RLS / SIN POLICIES por ahora
-- ======================================================

create extension if not exists pgcrypto;

-- =========================
-- CONVENIOS
-- =========================

create table if not exists public.convenios (
  id bigint generated by default as identity primary key,
  nombre text not null,
  documento text,
  celular text,
  direccion text,
  correo text,
  activo boolean not null default true,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create index if not exists convenios_nombre_idx
on public.convenios (upper(nombre));

alter table public.convenios disable row level security;

-- =========================
-- CONSECUTIVOS MATRÍCULA
-- =========================

create table if not exists public.consecutivos_matricula (
  anio integer primary key,
  ultimo integer not null default 0,
  updated_at timestamptz not null default now()
);

alter table public.consecutivos_matricula disable row level security;

-- =========================
-- APRENDICES
-- =========================

create table if not exists public.aprendices (
  id bigint generated by default as identity primary key,

  consecutivo text unique not null,
  fecha_matricula date not null default ((now() at time zone 'America/Bogota')::date),

  tipo_doc text,
  documento text,
  lugar_expedicion text,
  genero text,
  nombres text,
  apellidos text,
  fecha_nacimiento date,

  celular text,
  correo text,
  direccion text,
  barrio text,
  ciudad text,

  estado_civil text,
  ocupacion text,
  eps text,
  estrato integer,
  nivel_educativo text,

  convenio text,
  convenio_id bigint references public.convenios(id) on update cascade on delete set null,

  categorias text[],

  estado text not null default 'PENDIENTE'
    check (estado in ('PENDIENTE', 'ACTIVO', 'CERTIFICADO')),

  acudi_nombres text,
  acudi_apellidos text,
  acudi_tipo_doc text,
  acudi_documento text,
  acudi_celular text,
  acudi_direccion text,
  acudi_correo text,

  emergencia_nombre text,
  emergencia_celular text,

  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

alter table public.aprendices disable row level security;

create index if not exists aprendices_documento_idx
on public.aprendices (documento);

create index if not exists aprendices_fecha_matricula_idx
on public.aprendices (fecha_matricula);

create index if not exists aprendices_estado_idx
on public.aprendices (estado);

create index if not exists aprendices_convenio_id_idx
on public.aprendices (convenio_id);

-- =========================
-- FUNCIÓN CONSECUTIVO
-- =========================

create or replace function public.asignar_consecutivo_aprendiz()
returns trigger
language plpgsql
as $$
declare
  v_anio integer;
  v_siguiente integer;
begin
  if new.consecutivo is not null and new.consecutivo <> '' then
    return new;
  end if;

  v_anio := extract(year from now() at time zone 'America/Bogota');

  insert into public.consecutivos_matricula (anio, ultimo)
  values (v_anio, 0)
  on conflict (anio) do nothing;

  update public.consecutivos_matricula
  set ultimo = ultimo + 1,
      updated_at = now()
  where anio = v_anio
  returning ultimo into v_siguiente;

  new.consecutivo := v_anio::text || '-' || lpad(v_siguiente::text, 5, '0');

  return new;
end;
$$;

drop trigger if exists trg_asignar_consecutivo_aprendiz on public.aprendices;

create trigger trg_asignar_consecutivo_aprendiz
before insert on public.aprendices
for each row
execute function public.asignar_consecutivo_aprendiz();

2) Guarda también un archivo de verificación

Crea este archivo:

C:\mi-proyecto\supabase\migrations\001_verificacion_matriculas_convenios.sql

Con esto:

-- Verificación de tablas nuevas

select to_regclass('public.convenios') as tabla_convenios;
select to_regclass('public.consecutivos_matricula') as tabla_consecutivos_matricula;
select to_regclass('public.aprendices') as tabla_aprendices;

select count(*) as total_convenios
from public.convenios;

select column_name, data_type
from information_schema.columns
where table_schema = 'public'
  and table_name in ('convenios', 'consecutivos_matricula', 'aprendices')
order by table_name, ordinal_position;

select
  schemaname,
  tablename,
  rowsecurity
from pg_tables
where schemaname = 'public'
  and tablename in ('convenios', 'consecutivos_matricula', 'aprendices')
order by tablename;
3) Cómo aplicarlo después en producción

Cuando llegue el momento de pasarlo a la base real de App Script, ejecutas primero:

001_matriculas_convenios.sql

Luego ejecutas:

001_verificacion_matriculas_convenios.sql

Así confirmas que todo quedó creado.

4) Recomendación para organizar futuros cambios

Cada vez que creemos una tabla nueva o modifiquemos estructura, usa esta numeración:

001_matriculas_convenios.sql
002_caja.sql
003_runt_sicov_siet.sql
004_agendamiento.sql
005_reportes.sql

No sobrescribas archivos anteriores. Si luego hacemos cambios a aprendices, creamos otro archivo, por ejemplo:

006_alter_aprendices_campos_extra.sql

Eso te deja un historial limpio y seguro para producción.

5) Opcional: guardar en Git

Sí puedes subir estos SQL al repositorio, porque no contienen contraseñas.

Antes revisa:

cd C:\mi-proyecto
git status

Y cuando quieras respaldarlo:

git add supabase/migrations/001_matriculas_convenios.sql supabase/migrations/001_verificacion_matriculas_convenios.sql
git commit -m "Agregar migracion inicial de matriculas y convenios"

Con eso ya tienes el backup formal de las tablas nuevas.
